app_flask.py 
# ==============================================================================
# PROJETO ACADÉMICO: API BACKEND EM FLASK PARA MISTURAS QUÍMICAS
# Requisitos: pip install flask flask-cors
# Execução: python app_flask.py
# ==============================================================================

from flask import Flask, jsonify, request

app = Flask(__name__)

# Base de Dados
DATABASE_REACOES = [
    {
        "itens": ["agua_sanitaria", "alcool"],
        "status": "PERIGO",
        "titulo": "Formação de Clorofórmio Tóxico",
        "descricao": "Risco de desmaio, irritação pulmonar severa e danos hepáticos."
    },
    {
        "itens": ["agua_sanitaria", "limpa_vidros"],
        "status": "PERIGO",
        "titulo": "Libertação de Cloraminas",
        "descricao": "Gás irritante que causa queimaduras nas vias aéreas e asfixia."
    },
    {
        "itens": ["agua_sanitaria", "vinagre"],
        "status": "PERIGO",
        "titulo": "Gás Cloro Corrosivo",
        "descricao": "Gás altamente corrosivo e potencialmente letal em recintos fechados."
    }
]

@app.route('/api/avaliar-mistura', methods=['POST'])
def avaliar_mistura():
    data = request.get_json()
    p1 = data.get('produto1')
    p2 = data.get('produto2')
    
    req_set = set([p1, p2])
    
    for reacao in DATABASE_REACOES:
        if set(reacao["itens"]) == req_set:
            return jsonify({"sucesso": True, "resultado": reacao}), 200
            
    return jsonify({
        "sucesso": True, 
        "resultado": {
            "status": "SEGURO_OU_DESCONHECIDO",
            "titulo": "Sem Reação Perigosa Detetada",
            "descricao": "Mistura sem incompatibilidade crítica registada."
        }
    }), 200

if __name__ == '__main__':
    print("🚀 Servidor Flask de Segurança Química a rodar em http://localhost:5000")
    app.run(debug=True, port=5000)
